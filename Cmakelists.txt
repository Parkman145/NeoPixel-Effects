cmake_minimum_required(VERSION 4.3)
project(NEOFX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

FetchContent_Declare(CNDA GIT_REPOSITORY https://github.com/Parkman145/CNDA.git GIT_TAG main)

FetchContent_MakeAvailable(CNDA)

option (BUILD_EXE "Build main executable" ON)
option (BUILD_PYTHON "Build Python Library executable" ON)

# EXE ------------------------------
if(BUILD_EXE)
    # Find OpenCV
    find_package(OpenCV REQUIRED)
    message(STATUS "OpenCV version: ${OpenCV_VERSION}")
    # Add executable
    add_executable(main example/main.cpp)
    target_include_directories(main PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    # Link OpenCV libraries
    target_link_libraries(main 
        PRIVATE 
            ${OpenCV_LIBS}
            CNDA::CNDA
)
endif()

# Python ------------------------------
if(BUILD_PYTHON)
    add_subdirectory(extern/pybind11)
    pybind11_add_module(FXLIB bindings/bindings.cpp)
    target_include_directories(FXLIB PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

    target_link_libraries(FXLIB PRIVATE CNDA::CNDA)
endif()
